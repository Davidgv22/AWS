import pymysql 

import json 

import os 

 

# Configuración mediante variables de entorno (Requisito 3.2) 

DB_HOST = os.environ.get('DB_HOST') 

DB_USER = os.environ.get('DB_USER') 

DB_PASS = os.environ.get('DB_PASS') 

DB_NAME = os.environ.get('DB_NAME') 

 

def lambda_handler(event, context): 

    connection = None 

    try: 

        # 1. Manejo del Body (ReqBin envía los datos como string en event['body']) 

        if 'body' in event: 

            # Si el body es string, lo convertimos a diccionario 

            data = json.loads(event['body']) if isinstance(event['body'], str) else event['body'] 

        else: 

            # Si pruebas desde el botón "Test" de la consola de AWS 

            data = event 

 

        # 2. Extracción de datos del JSON 

        user_id = data.get('id') 

        nombre  = data.get('nombre') 

        apellido = data.get('apellido', '') # Valor por defecto vacío si no se envía 

        telefono = data.get('telefono', '') 

        gmail    = data.get('gmail', '') 

        ciudad   = data.get('ciudad', '') 

 

        # Validamos que al menos id y nombre existan 

        if not user_id or not nombre: 

            return { 

                'statusCode': 400, 

                'body': json.dumps({'error': 'Faltan campos obligatorios: id y nombre'}) 

            } 

 

        # 3. Conexión a RDS 

        connection = pymysql.connect( 

            host=DB_HOST, 

            user=DB_USER, 

            password=DB_PASS, 

            database=DB_NAME, 

            cursorclass=pymysql.cursors.DictCursor, 

            connect_timeout=5 

        ) 

 

        with connection.cursor() as cursor: 

            # 4. Ejecución del INSERT (Asegúrate de que coincida con tus columnas) 

            sql = """INSERT INTO personas (id, nombre, apellido, telefono, gmail, ciudad)  

                     VALUES (%s, %s, %s, %s, %s, %s)""" 

            cursor.execute(sql, (user_id, nombre, apellido, telefono, gmail, ciudad)) 

             

        connection.commit() 

 

        return { 

            'statusCode': 201, 

            'headers': { 

                'Content-Type': 'application/json', 

                'Access-Control-Allow-Origin': '*'  # Permite peticiones desde ReqBin/Navegador 

            }, 

            'body': json.dumps({ 

                'mensaje': '¡Logrado! Registro insertado con éxito', 

                'id_insertado': user_id, 

                'nombre': nombre 

            }) 

        } 

 

    except Exception as e: 

        print(f"Error detallado: {e}") 

        return { 

            'statusCode': 500, 

            'body': json.dumps({'error': str(e)}) 

        } 

 

    finally: 

        if connection and connection.open: 

            connection.close()
